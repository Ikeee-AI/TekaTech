## Reproducing the scoped run
The scoped answers come from this local script, not committed as a tool because it hardcodes the two TekaTech-dev gadget ids and the repo path. It imports `scripts/eval_adapter.py` unchanged and only adds `gadget_ids` to each `POST /query`, the way the gadget and compare pages do. Run it against a local backend on TekaTech-dev: `python scoped_run.py <out.json>`, then score the output with `scripts/eval_runner.py`.

```python
"""Local, uncommitted: the adapter's own collect() with gadget_ids added to each POST /query (#88)."""
import importlib.util, json, sys
from datetime import datetime, timezone
from pathlib import Path
ROOT = Path("D:/SchoolPurposes/AI/endtermgroupproject/TekaTech")
spec = importlib.util.spec_from_file_location("eval_adapter", ROOT / "scripts" / "eval_adapter.py")
ad = importlib.util.module_from_spec(spec); spec.loader.exec_module(ad)

API = "http://127.0.0.1:8000"
IDS = {"iphone": "3882d510-da5b-4a56-ae58-4b5c165aae31", "galaxy": "1d4e868c-e0b0-4c30-aaaa-140e12713eae"}
def scope(q):
    q = q.lower(); ids = []
    if "iphone" in q: ids.append(IDS["iphone"])
    if "galaxy" in q or "s24" in q: ids.append(IDS["galaxy"])
    return ids
used = {}
def http(method, url, body=None):
    if method == "POST" and url.endswith("/query"):
        ids = scope(body["question"]); used[body["question"]] = ids
        if ids: body = {**body, "gadget_ids": ids}
    return ad.http_json(method, url, body)

cases = json.loads((ROOT / "prompts/eval/cases.json").read_text(encoding="utf-8"))["cases"]
snippets = json.loads((ROOT / "prompts/eval/sources.json").read_text(encoding="utf-8"))["snippets"]
missing, merged = ad.check_pack(ad.stored_chunks(API, http), snippets)
assert not missing and not merged, (missing, merged)
print("pack check passed; scopes:", {c["id"]: len(scope(c["query"])) for c in cases}, flush=True)
answers, removed, errors, models = ad.collect(API, cases, snippets, http)
out = {"provider": f"TekaTech POST /query ({API}), gadget-scoped", "model": ", ".join(sorted(m for m in models if m)),
       "prompt_version": ad.default_prompt_version(), "origin": "live",
       "generated_at": datetime.now(timezone.utc).isoformat(),
       "note": "Real /query output with gadget_ids sent per case (branch tekatech/s4-ma-scoped-retrieval, #88). Citation markers rewritten from retrieval position to pack ids. invalid_citations_removed counts markers the backend stripped; score them as invalid.",
       "invalid_citations_removed": removed, "errors": errors, "answers": answers}
Path(sys.argv[1]).write_text(json.dumps(out, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")
print(f"wrote {len(answers)} answers, {len(errors)} errors, {sum(removed.values())} removed")
```
